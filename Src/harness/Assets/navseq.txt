#                     anyway, so redundant here; kept for explicitness.
#   delay=<seconds>   wait this long before the NEXT url, and before each one after it until changed.
#                     0 means "as soon as possible" -- fires a new nav on top of a still-loading one.
#   <url>             navigated to through the same code path a finger on the address bar would take.
